package cn.zero3d.buglywebrequest;

import com.tencent.bugly.network.ClientBuilder;
import java.io.*;
import java.net.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicBoolean;
import okhttp3.*;

/** One independently retried logical request per URL; exactly one final callback. */
public final class WireBridge {
    public interface Callback { void onComplete(int index, Result result); }
    private static final ExecutorService workers = Executors.newFixedThreadPool(8, task -> {
        Thread thread = new Thread(task, "BuglyWebRequest"); thread.setDaemon(true); return thread;
    });
    private static final ScheduledExecutorService timers = Executors.newScheduledThreadPool(8, task -> {
        Thread thread = new Thread(task, "BuglyWebRequest-timeout"); thread.setDaemon(true); return thread;
    });
    private static final OkHttpClient baseClient = new OkHttpClient.Builder().retryOnConnectionFailure(false).build();
    private static final class Attempt {
        final long started = System.nanoTime();
        final AtomicBoolean timedOut = new AtomicBoolean();
        volatile Call call;
        volatile HttpURLConnection connection;
        void close() { if (call != null) call.cancel(); if (connection != null) connection.disconnect(); }
    }

    public static final class Result {
        private byte[] data = new byte[0];
        private int status, attempts;
        private long durationMs, lastAttemptMs;
        private boolean success;
        private String failure = "", url;
        public byte[] getData() { return data; }
        public int getStatus() { return status; }
        public int getAttempts() { return attempts; }
        public long getDurationMs() { return durationMs; }
        public long getLastAttemptMs() { return lastAttemptMs; }
        public boolean isSuccess() { return success; }
        public String getFailure() { return failure; }
        public String getUrl() { return url; }
    }

    public static final class Job {
        private final AtomicBoolean finished = new AtomicBoolean();
        private final AtomicBoolean canceled = new AtomicBoolean();
        private final int index;
        private final String url;
        private final Callback callback;
        private volatile long started;
        private volatile int attempts;
        private volatile Attempt active;
        private volatile Thread worker;
        private Job(int index, String url, Callback callback) { this.index=index; this.url=url; this.callback=callback; }
        private void closeAttempt() {
            Attempt current=active;
            if (current != null) current.close();
        }
        public void cancel() {
            if (finished.get()) return;
            canceled.set(true);
            Result result = new Result(); result.url=url; result.failure="canceled"; result.attempts=attempts;
            Attempt current = active;
            if (current != null) result.lastAttemptMs=elapsed(current.started);
            finish(result);
            closeAttempt();
            Thread running = worker;
            if (running != null) running.interrupt();
        }
        private void finish(Result result) {
            if (!finished.compareAndSet(false, true)) return;
            result.durationMs = started == 0 ? 0 : elapsed(started);
            callback.onComplete(index, result);
        }
    }

    private static long elapsed(long start) { return Math.max(0, Math.round((System.nanoTime()-start)/1000000.0)); }
    private static byte[] read(InputStream stream, int limit) throws IOException {
        try (InputStream input=stream; ByteArrayOutputStream output=new ByteArrayOutputStream()) {
            if (input != null) {
                byte[] buffer=new byte[8192]; int count;
                while ((count=input.read(buffer)) != -1) {
                    if ((long)output.size()+count > limit) throw new ResponseTooLarge();
                    output.write(buffer,0,count);
                }
            }
            return output.toByteArray();
        }
    }
    private static final class ResponseTooLarge extends IOException { }

    public static Job submit(int index, String url, byte[] body, String[] names, String[] values,
        int transport, int timeoutSeconds, int maxAttempts, int retryDelayMs, int maxResponseBytes,
        boolean monitor, Callback callback) {
        Job job=new Job(index,url,callback);
        workers.execute(() -> run(job,body,names,values,transport,timeoutSeconds,maxAttempts,retryDelayMs,maxResponseBytes,monitor));
        return job;
    }

    private static void run(Job job, byte[] body, String[] names, String[] values, int transport,
        int timeoutSeconds, int maxAttempts, int retryDelayMs, int limit, boolean monitor) {
        job.worker=Thread.currentThread();
        if (job.finished.get()) { job.worker=null; return; }
        job.started=System.nanoTime();
        Result terminal=new Result(); terminal.url=job.url;
        try {
            URL target=new URL(job.url);
            if (!"http".equalsIgnoreCase(target.getProtocol()) && !"https".equalsIgnoreCase(target.getProtocol()))
                throw new IllegalArgumentException();
            for (int attempt=1; attempt<=maxAttempts && !job.canceled.get(); attempt++) {
                Result result=new Result(); result.url=job.url; result.attempts=attempt; job.attempts=attempt;
                Attempt control=new Attempt(); job.active=control;
                long attemptStart=System.nanoTime();
                ScheduledFuture<?> timeout=timers.schedule(() -> { control.timedOut.set(true); control.close(); }, timeoutSeconds, TimeUnit.SECONDS);
                boolean retry=false;
                try {
                    if (transport == 0) {
                        OkHttpClient.Builder builder=baseClient.newBuilder().callTimeout(timeoutSeconds,TimeUnit.SECONDS)
                            .connectTimeout(timeoutSeconds,TimeUnit.SECONDS).readTimeout(timeoutSeconds,TimeUnit.SECONDS)
                            .writeTimeout(timeoutSeconds,TimeUnit.SECONDS);
                        if (monitor) builder=ClientBuilder.configBuilder(builder);
                        Request.Builder request=new Request.Builder().url(job.url);
                        for (int i=0;i<names.length;i++) request.header(names[i],values[i]);
                        control.call=builder.build().newCall(request.post(RequestBody.create(null,body)).build());
                        if (job.canceled.get() || control.timedOut.get()) control.call.cancel();
                        try (Response response=control.call.execute()) {
                            result.status=response.code();
                            result.data=read(response.body()==null?null:response.body().byteStream(),limit);
                        }
                    } else {
                        HttpURLConnection connection=(HttpURLConnection)target.openConnection();
                        control.connection=connection;
                        if (monitor) { connection=(HttpURLConnection)ClientBuilder.wrapURLConnection(connection); control.connection=connection; }
                        if (job.canceled.get() || control.timedOut.get()) throw new InterruptedIOException();
                        connection.setRequestMethod("POST"); connection.setDoOutput(true); connection.setUseCaches(false);
                        int milliseconds=(int)Math.min(Integer.MAX_VALUE,(long)timeoutSeconds*1000);
                        connection.setConnectTimeout(milliseconds); connection.setReadTimeout(milliseconds);
                        for (int i=0;i<names.length;i++) connection.setRequestProperty(names[i],values[i]);
                        try (OutputStream output=connection.getOutputStream()) { output.write(body); }
                        result.status=connection.getResponseCode();
                        result.data=read(result.status>=400?connection.getErrorStream():connection.getInputStream(),limit);
                    }
                    if (control.timedOut.get()) throw new SocketTimeoutException();
                    result.success=result.status>=200 && result.status<300;
                    if (!result.success) result.failure="http_"+result.status;
                    retry=result.status==408 || result.status==429 || result.status>=500;
                } catch (Exception error) {
                    if (job.canceled.get()) result.failure="canceled";
                    else if (control.timedOut.get() || error instanceof SocketTimeoutException || error instanceof InterruptedIOException) result.failure="timeout";
                    else if (error instanceof ResponseTooLarge) result.failure="response_too_large";
                    else result.failure="connection_error:"+error.getClass().getSimpleName();
                    retry=!job.canceled.get() && !(error instanceof ResponseTooLarge) && !(error instanceof javax.net.ssl.SSLException);
                } finally {
                    timeout.cancel(false); result.lastAttemptMs=elapsed(attemptStart);
                    if (control.connection!=null) control.connection.disconnect();
                    job.active=null;
                }
                terminal=result;
                if (result.success || !retry || attempt==maxAttempts) break;
                Thread.sleep(Math.min(2000L,(long)retryDelayMs*attempt));
            }
        } catch (InterruptedException ignored) { terminal.failure="canceled"; terminal.success=false; }
        catch (Exception ignored) { terminal.failure="invalid_url"; terminal.success=false; }
        finally {
            if (job.canceled.get()) { terminal.failure="canceled"; terminal.success=false; }
            job.finish(terminal); job.worker=null;
        }
    }
}
